% Slide 1
\begin{frame}{Targeting the top 20\% could materially improve telephone-campaign productivity}
\vspace{-0.2em}
\begin{block}{Recommendation}
Replace broad customer selection with a \textbf{pre-campaign logistic ranking}, then pilot the policy before full rollout.
\end{block}

\vspace{0.7em}
\begin{columns}[T,totalwidth=\textwidth]
\begin{column}{0.31\textwidth}
\centering
\metric{49.7\%}{top 20\% conversion}
\end{column}
\begin{column}{0.31\textwidth}
\centering
\metric{30.8\%}{test-period random baseline}
\end{column}
\begin{column}{0.31\textwidth}
\centering
\metric{+310.88}{incremental subscriptions}
\end{column}
\end{columns}

\vspace{1.0em}
\begin{itemize}\tightitem
  \item In the ordered test period, the top 20\% of scored customers converted at \textbf{49.7\%} versus \textbf{30.8\%} under random selection.
  \item This is a \textbf{1.61$\times$ lift} across \textbf{1,648 calls}.
  \item Use the ranking as a controlled pilot, not as immediate full-scale automation.
\end{itemize}
\end{frame}

% Slide 2
\begin{frame}{The decision is whom to call first -- not who appears likely after the call}
\begin{columns}[T,totalwidth=\textwidth]
\begin{column}{0.55\textwidth}
\begin{itemize}\tightitem
  \item \textbf{Objective:} maximize subscriptions within fixed telephone capacity.
  \item \textbf{Scoring moment:} before current-campaign contact.
  \item \textbf{Valid inputs:} customer profile, financial relationship, prior-campaign history, and known macro context.
  \item \textbf{Primary metrics:} conversion rate and lift at capacity cuts.
\end{itemize}
\end{column}
\begin{column}{0.42\textwidth}
\begin{block}{Excluded from strict model}
\begin{itemize}\tightitem
  \item \texttt{duration}: known after the call.
  \item \texttt{campaign}: generated during current campaign.
  \item \texttt{contact}, \texttt{month}, \texttt{day\_of\_week}: current scheduling/channel fields.
\end{itemize}
\end{block}
\end{column}
\end{columns}

% Slide 3
\begin{frame}{The data supports ranking, but quality limits and temporal drift constrain confidence}
\begin{columns}[T,totalwidth=\textwidth]
\begin{column}{0.57\textwidth}
\begin{itemize}\tightitem
  \item \textbf{41,188 contact records}; 12 exact duplicates retained for contact-opportunity analysis.
  \item No stable customer ID: records are not necessarily unique customers.
  \item Prior success converted at \textbf{65.11\%}; prior contact within 0--7 days at \textbf{65.76\%}.
  \item Low-Euribor quartile converted at \textbf{25.38\%} versus \textbf{5.08\%} in the third quartile.
  \item More current-campaign contacts coincide with lower conversion, \textbf{descriptively, not causally}.
  \item \texttt{duration} shows severe post-contact leakage and is not actionable for selection.
\end{itemize}
\end{column}
\begin{column}{0.40\textwidth}
\begin{block}{Regime shift is material}
\centering
\metric{6.37\%}{train conversion}\quad
\metric{30.83\%}{test conversion}

\vspace{0.7em}
\small Source-order quintiles rose from \textbf{3.14\%} to \textbf{30.84\%}.
\end{block}
\vspace{0.4em}
\small Literal \texttt{unknown} values remain categories; \texttt{pdays=999} means no prior contact, not missingness.
\end{column}
\end{columns}

\vspace{0.5em}
\begin{block}{Interpretation: ordered holdout is deliberately conservative}
Source order is an \textbf{approximate temporal validation}, not exact chronology. The shift supports forward-style testing and makes prospective validation mandatory.
\end{block}
\end{frame}

\vspace{0.25em}
\begin{block}{Why logistic}
Boosting improves ROC-AUC, but logistic has better PR-AUC, Brier score, and top-10\% performance. All lifts use the \textbf{30.83\% contemporaneous test baseline}; overall discrimination is modest.
\end{block}

% Slide 5
\begin{frame}{A controlled top-20\% pilot should test value, safety, and transportability}
\begin{columns}[T,totalwidth=\textwidth]
\begin{column}{0.48\textwidth}
\begin{block}{Pilot design}
\begin{enumerate}\tightitem
  \item Apply consent, suppression, legal, and deduplication rules; then score before campaign launch.
  \item Randomize customers near or across the cutoff:
    \begin{itemize}\tightitem
      \item scored targeting group;
      \item business-as-usual or random control.
    \end{itemize}
  \item Measure conversions per call, deposits and margin, call cost and agent time, and complaints or opt-outs.
  \item Predefine sample size, stopping rules, guardrails, and rollout criteria; monitor score and calibration drift.
\end{enumerate}
\end{block}
\end{column}
\begin{column}{0.49\textwidth}
\begin{block}{Risks and limitations}
\begin{itemize}\tightitem
  \item Prediction is not causal uplift.
  \item No stable customer identifier.
  \item Source order is only approximate chronology.
  \item Train conversion was 6.37\%; test conversion was 30.83\%.
  \item Historical campaign selection may bias labels.
  \item No deposit amount, margin, or call-cost fields.
  \item Raw scores are not economic values; capacity cutoffs may drift.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\vspace{0.5em}
\centering\small Human owner approves cutoff, eligibility, risk controls, experiment design, and any rollout.
\end{frame}
