% Slide 1
\begin{frame}{Targeting the top 20\% could materially improve telephone-campaign productivity}
\vspace{-0.2em}
\begin{block}{Recommendation}
Replace broad customer selection with a \textbf{pre-campaign logistic ranking}, then pilot the policy before full rollout.
\end{block}

\vspace{0.7em}
\begin{columns}[T,totalwidth=\textwidth]
\begin{column}{0.31\textwidth}
\centering
\metric{49.7\%}{top 20\% conversion}
\end{column}
\begin{column}{0.31\textwidth}
\centering
\metric{30.8\%}{test-period random baseline}
\end{column}
\begin{column}{0.31\textwidth}
\centering
\metric{+311}{incremental subscriptions}
\end{column}
\end{columns}

\vspace{1.0em}
\begin{itemize}\tightitem
  \item In the ordered test period, the top 20\% of scored customers converted at \textbf{49.7\%} versus \textbf{30.8\%} under random selection.
  \item This is a \textbf{1.61$\times$ lift} across \textbf{1,648 calls}.
  \item Use the ranking as a controlled pilot, not as immediate full-scale automation.
\end{itemize}
\end{frame}

% Slide 2
\begin{frame}{The decision is whom to call first -- not who appears likely after the call}
\begin{columns}[T,totalwidth=\textwidth]
\begin{column}{0.55\textwidth}
\begin{itemize}\tightitem
  \item \textbf{Objective:} maximize subscriptions within fixed telephone capacity.
  \item \textbf{Scoring moment:} before current-campaign contact.
  \item \textbf{Valid inputs:} customer profile, financial relationship, prior-campaign history, and known macro context.
  \item \textbf{Primary metrics:} conversion rate and lift at capacity cuts.
\end{itemize}
\end{column}
\begin{column}{0.42\textwidth}
\begin{block}{Excluded from strict model}
\begin{itemize}\tightitem
  \item \texttt{duration}: known after the call.
  \item \texttt{campaign}: generated during current campaign.
  \item \texttt{contact}, \texttt{month}, \texttt{day\_of\_week}: current scheduling/channel fields.
\end{itemize}
\end{block}
\end{column}
\end{columns}

% Slide 5
\begin{frame}{Deploy through a controlled pilot, not immediate full-scale automation}
\begin{columns}[T,totalwidth=\textwidth]
\begin{column}{0.48\textwidth}
\begin{block}{Pilot design}
\begin{enumerate}\tightitem
  \item Score eligible customers before campaign launch.
  \item Randomize customers near or across the cutoff:
    \begin{itemize}\tightitem
      \item scored targeting group;
      \item business-as-usual or random control.
    \end{itemize}
  \item Measure conversions per call, deposits and margin, agent time, and complaints or opt-outs.
  \item Retrain and recalibrate as campaign conditions change.
\end{enumerate}
\end{block}
\end{column}
\begin{column}{0.49\textwidth}
\begin{block}{Risks and limitations}
\begin{itemize}\tightitem
  \item Prediction is not causal uplift.
  \item No stable customer identifier.
  \item Source order is only approximate chronology.
  \item Train conversion was 6.37\%; test conversion was 30.83\%.
  \item Historical campaign selection may bias labels.
  \item No deposit amount, margin, or call-cost fields.
  \item Probabilities are not calibrated enough for direct economic optimization.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}
